163def _disputa_resumen(reserva): 

164 """Disputa asociada a la reserva, si existe (se serializa al vuelo)."""

171def validar_evidencia(archivo): 

172 """Valida tipo y tamaño de un archivo de evidencia (ticket 5.2).""" 

173 if archivo.size > MAX_EVIDENCIA_BYTES: 

174 raise serializers.ValidationError("El archivo supera el máximo de 5 MB.") 

175 if archivo.content_type and archivo.content_type not in TIPOS_EVIDENCIA: 

176 raise serializers.ValidationError("Solo se aceptan PDF, JPG o PNG.") 

177 return archivo

279 resolucion = serializers.CharField(max_length=2000)

284 def validate(self, attrs): 

285 if attrs["resultado"] == Disputa.RESULTADO_DIVISION: 

286 if not attrs.get("porcentaje_reembolso"): 

287 raise serializers.ValidationError( 

288 {"porcentaje_reembolso": "Indica el porcentaje de la división."} 

289 ) 

290 else: 

291 attrs.pop("porcentaje_reembolso", None) 

292 return attrs
